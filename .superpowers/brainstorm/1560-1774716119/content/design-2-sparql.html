# ① 根据症状关键词匹配诊断规则
def query_matching_rules(keywords: List[str]) → List[ActivatedNode]: """ SPARQL: 在 rules_model.ttl 中找与关键词匹配的规则 返回规则ID、中文描述、触发条件 """
# ② 根据信号值定位 Ontology 个体
def query_signal_individuals(signals: Dict[str,str]) → Dict[str,str]: """ SPARQL: sv-pm="0:Off" → :PowerMode_Off (个体) sv-kv="INVALID" → :KeyValidStatus_Invalid 信号值与本体个体的精确映射 """
# ③ 获取规则的完整推理链
def query_rule_chain(rule_id: str) → str: """ SPARQL: 给定规则ID，返回 前提条件 → 结论 的完整三元组链 供 LLM prompt 和前端溯源展示使用 """
